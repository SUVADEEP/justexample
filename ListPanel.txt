/**
 * Data class representing a destination/page with its composable function
 * This allows registering destinations with their associated composables
 */
data class AppDestination @OptIn(ExperimentalMaterial3AdaptiveApi::class) constructor(
    val id: String,
    val label: String,
    val icon: ImageVector,
    val contentDescription: String,
    val role: ThreePaneScaffoldRole,
    val composable: @Composable (AppDestination, ThreePaneScaffoldNavigator<AppDestination>, CoroutineScope) -> Unit  // Composable function to render the page
) {
    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (javaClass != other?.javaClass) return false
        other as AppDestination
        return id == other.id
    }

    override fun hashCode(): Int {
        return id.hashCode()
    }
}

/**
 * Registry for all app destinations with their composables
 * Destinations can be registered dynamically from anywhere in the app
 */
object AppDestinationRegistry {
    // Use mutable list so we can add/remove destinations at runtime
    private val _destinations = mutableListOf<AppDestination>()
    
    // Public read-only access to destinations
    val destinations: List<AppDestination>
        get() = _destinations.toList()  // Return a copy to prevent external modification
    
    init {
        // Register default destinations
        registerDefaultDestinations()
    }
    
    /**
     * Register default destinations
     */
    @OptIn(ExperimentalMaterial3AdaptiveApi::class)
    private fun registerDefaultDestinations() {
        _destinations.addAll(
            listOf(
                AppDestination(
                    id = "home",
                    label = "Home",
                    icon = Icons.Default.Home,
                    contentDescription = "Home",
                    role = ListDetailPaneScaffoldRole.Detail,
                    composable = { destination, navigator, scope -> HomeDestination(destination, navigator, scope) }
                ),
                AppDestination(
                    id = "favorites",
                    label = "Fav",
                    icon = Icons.Default.Favorite,
                    contentDescription = "Fav",
                    role = ListDetailPaneScaffoldRole.Detail,
                    composable = { destination, navigator, scope -> FavoritesDestination(destination, navigator, scope) }
                ),
                AppDestination(
                    id = "shopping",
                    label = "Shopping",
                    icon = Icons.Default.ShoppingCart,
                    contentDescription = "Shopping",
                    role = ListDetailPaneScaffoldRole.Detail,
                    composable = { destination, navigator, scope -> ShoppingDestination(destination, navigator, scope) }
                ),
                AppDestination(
                    id = "profile",
                    label = "Profile",
                    icon = Icons.Default.AccountBox,
                    contentDescription = "Profile",
                    role = ListDetailPaneScaffoldRole.Detail,
                    composable = { destination, navigator, scope -> ProfileDestination(destination, navigator, scope) }
                ),
                AppDestination(
                    id = "extra",
                    label = "Extra",
                    icon = Icons.Default.AccountBox,
                    contentDescription = "Extra",
                    role = ListDetailPaneScaffoldRole.Extra,
                    composable = { destination, navigator, scope -> Extra(destination, navigator, scope) }
                ),
            )
        )
    }
    
    /**
     * Register a new destination (can be called from anywhere)
     * @param destination The destination to register
     * @return true if registered successfully, false if destination with same id already exists
     */
    fun register(destination: AppDestination): Boolean {
        if (_destinations.any { it.id == destination.id }) {
            return false  // Destination with this id already exists
        }
        _destinations.add(destination)
        return true
    }
    
    /**
     * Register multiple destinations at once
     */
    fun registerAll(destinations: List<AppDestination>) {
        destinations.forEach { register(it) }
    }
    
    /**
     * Unregister a destination by id
     * @param id The id of the destination to remove
     * @return true if removed, false if not found
     */
    fun unregister(id: String): Boolean {
        return _destinations.removeIf { it.id == id }
    }
    
    /**
     * Clear all destinations
     */
    fun clear() {
        _destinations.clear()
    }
    
    /**
     * Check if a destination with given id exists
     */
    fun exists(id: String): Boolean {
        return _destinations.any { it.id == id }
    }
    
    /**
     * Find destination by id
     */
    fun findById(id: String): AppDestination? {
        return _destinations.firstOrNull { it.id == id }
    }
    
    /**
     * Get destinations by role
     */
    fun getByRole(role: ThreePaneScaffoldRole): List<AppDestination> {
        return _destinations.filter { it.role == role }
    }
}


// Pane expansion anchors for drag functionality
private val PaneExpansionAnchors = listOf(
    PaneExpansionAnchor.Proportion(0.3f),
    PaneExpansionAnchor.Proportion(0.5f),
    PaneExpansionAnchor.Proportion(0.7f),
)


@Composable
fun PaneExpansionDragHandleSample(
    state: PaneExpansionState
) {
    val interactionSource = remember { MutableInteractionSource() }
    VerticalDragHandle(
        modifier = Modifier,
        interactionSource = interactionSource,
    )
}

@OptIn(ExperimentalMaterial3AdaptiveApi::class)
@Composable
fun NavigationSample(){

    val scaffoldNavigator = rememberListDetailPaneScaffoldNavigator<AppDestination>()
    val scope = rememberCoroutineScope()
    val backNavigationBehavior = BackNavigationBehavior.PopUntilScaffoldValueChange
    
    // Remember the last Detail destination to keep it visible when Extra is active
    var lastDetailDestination by remember { 
        mutableStateOf<AppDestination?>(null) 
    }
    
    // Update lastDetailDestination when a Detail destination is active
    val currentDestination = scaffoldNavigator.currentDestination?.contentKey
    if (currentDestination?.role == ListDetailPaneScaffoldRole.Detail) {
        lastDetailDestination = currentDestination
    }

    NavigableListDetailPaneScaffold(
        paneExpansionState =
            rememberPaneExpansionState(
                keyProvider = scaffoldNavigator.scaffoldValue,
                anchors = PaneExpansionAnchors,
                initialAnchoredIndex = 1,
            ),
        paneExpansionDragHandle = { state ->
            PaneExpansionDragHandleSample(state)
        },
        navigator = scaffoldNavigator,
        listPane = {
            AnimatedPane {
                Column(
                    modifier = Modifier.fillMaxSize()
                ) {
                    Text(
                        text = "Main Pane",
                        style = MaterialTheme.typography.displayMedium,
                        modifier = Modifier.padding(16.dp)
                    )
                    MyList(
                        onItemClick = { item ->
                            // Navigate to the detail pane with the passed item
                            scope.launch {
                                scaffoldNavigator.navigateTo(
                                    item.role,
                                    item
                                )
                            }
                        },
                    )

                }

            }
        },
        detailPane = {
            AnimatedPane {
                // Show detail pane content: current Detail destination, or last Detail destination when Extra is active
                // This ensures the list pane remains visible on larger screens
                val detailDestination = currentDestination
                    ?.takeIf { it.role == ListDetailPaneScaffoldRole.Detail }
                    ?: lastDetailDestination
                
                detailDestination?.let { contentKey ->
                    Column {
                        // Allow users to dismiss the detail pane. Use back navigation to
                        // hide an expanded detail pane.
                        if (scaffoldNavigator.scaffoldValue[ListDetailPaneScaffoldRole.Detail] == PaneAdaptedValue.Expanded) {
                            // Material design principles promote the usage of a right-aligned
                            // close (X) button.
                            IconButton(
                                modifier =  Modifier.align(Alignment.End).padding(16.dp),
                                onClick = {
                                    scope.launch {
                                        scaffoldNavigator.navigateBack(backNavigationBehavior)
                                    }
                                }
                            ) {
                                Icon(Icons.Default.Close, contentDescription = "Close")
                            }
                        }
                        IconButton(
                            onClick = {
                                scope.launch {
                                    // Find EXTRA destination from registry
                                    val extraDestination = AppDestinationRegistry.findById("extra")
                                    extraDestination?.let {
                                        scaffoldNavigator.navigateTo(
                                            ListDetailPaneScaffoldRole.Extra,
                                            it
                                        )
                                    }
                                }
                            }
                        ) {
                            Icon(Icons.Default.AccountTree, contentDescription = "Extra")
                        }
                        // Call the composable function registered with this destination
                        contentKey.composable(contentKey, scaffoldNavigator, scope)
                    }
                }
            }
        },

        extraPane = {
            AnimatedPane {
                Column {

                        IconButton(
                            modifier =  Modifier.align(Alignment.End).padding(16.dp),
                            onClick = {
                                scope.launch {
                                    scaffoldNavigator.navigateBack(backNavigationBehavior)
                                }
                            }
                        ) {
                            Icon(Icons.Default.Close, contentDescription = "Close")
                        }

                    scaffoldNavigator.currentDestination?.contentKey
                        ?.takeIf { it.role == ListDetailPaneScaffoldRole.Extra }
                        ?.let { contentKey ->
                            // Call the composable function registered with this destination
                            (contentKey as AppDestination).composable(contentKey, scaffoldNavigator, scope)
                        }
                }
            }
        }
    )

}

@Composable
fun MyList(onItemClick: (AppDestination) -> Unit) {
    // Get destinations from configurable registry
    val destinations = AppDestinationRegistry.destinations
    
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp)
    ) {
        Text(
            text = "Settings",
            style = MaterialTheme.typography.headlineMedium,
            modifier = Modifier.padding(bottom = 16.dp)
        )
        
        LazyColumn {
            items(
                items = destinations,
                key = { it.id }  // Use id instead of name
            ) { destination ->
                Card(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = 4.dp)
                        .clickable { onItemClick(destination) },
                    colors = CardDefaults.cardColors(
                        containerColor = MaterialTheme.colorScheme.surface
                    ),
                    elevation = CardDefaults.cardElevation(
                        defaultElevation = 2.dp
                    )
                ) {
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(16.dp)
                    ) {
                        Icon(
                            imageVector = destination.icon,
                            contentDescription = destination.contentDescription,
                            modifier = Modifier.size(24.dp),
                            tint = MaterialTheme.colorScheme.primary
                        )
                        Spacer(modifier = Modifier.padding(horizontal = 12.dp))
                        Text(
                            text = destination.label,
                            style = MaterialTheme.typography.titleMedium,
                            color = MaterialTheme.colorScheme.onSurface
                        )


                    }
                }
            }
        }
    }
}
